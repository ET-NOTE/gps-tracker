# /etc/systemd/system/gps-pdf-render.socket
# Written by ops/pdf_service.py. Remove with `pdf_service.py rollback`.
[Unit]
Description=GPS PDF processor: page requests from the API
# No socket without a passing proof, and a new proof whenever the proof is run again.
Requires=gps-pdf-budget.service gps-pdf-render-selftest.service
After=gps-pdf-budget.service gps-pdf-render-selftest.service
PartOf=gps-pdf-render-selftest.service
# Not an early-boot socket: it waits for the proof, an ordinary service that runs for seconds.
# With the default dependencies it would have to come before sockets.target and after it at once.
DefaultDependencies=no
Conflicts=shutdown.target
Before=shutdown.target
[Socket]
ListenStream=/run/gps-pdf-render/render.sock
Accept=yes
SocketUser=root
SocketGroup=gps-api
SocketMode=0660
DirectoryMode=0755
MaxConnections=4
RemoveOnStop=yes
[Install]
WantedBy=multi-user.target
