cmake_minimum_required(VERSION 3.16)
# A profile is mandatory: never reuse the historical build/caltest.bin implicitly.
if(NOT FIRMWARE_PROFILE MATCHES "^(operating|https-operating|https-pilot|https-sleep-test|tls-cert-test|kc|battery)$")
  message(FATAL_ERROR "Select operating, https-operating, https-pilot, https-sleep-test, tls-cert-test, kc or battery")
endif()
if(FIRMWARE_PROFILE STREQUAL "https-operating" OR FIRMWARE_PROFILE STREQUAL "https-pilot" OR FIRMWARE_PROFILE STREQUAL "https-sleep-test" OR FIRMWARE_PROFILE STREQUAL "tls-cert-test")
  add_compile_definitions(GPS_HTTPS_PILOT=1)
endif()
if(FIRMWARE_PROFILE STREQUAL "https-operating")
  add_compile_definitions(GPS_HTTPS_OPERATING=1)
endif()
if(FIRMWARE_PROFILE STREQUAL "https-sleep-test")
  add_compile_definitions(GPS_HTTPS_SLEEP_TEST=1)
endif()
if(FIRMWARE_PROFILE STREQUAL "tls-cert-test")
  if(NOT EXISTS "${TLS_TEST_CA_HEADER}")
    message(FATAL_ERROR "TLS_TEST_CA_HEADER must point to the generated test-only CA header")
  endif()
  add_compile_definitions(GPS_TLS_CERT_TEST=1 GPS_TLS_TEST_CA_HEADER="${TLS_TEST_CA_HEADER}")
endif()
get_filename_component(EXPECTED_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/build/${FIRMWARE_PROFILE}" ABSOLUTE)
if(NOT CMAKE_BINARY_DIR STREQUAL EXPECTED_BUILD)
  message(FATAL_ERROR "Use the separate build directory: ${EXPECTED_BUILD}")
endif()
if(NOT SDKCONFIG STREQUAL "${EXPECTED_BUILD}/sdkconfig")
  message(FATAL_ERROR "SDKCONFIG must be isolated in ${EXPECTED_BUILD}/sdkconfig")
endif()
if(NOT FIRMWARE_REVISION MATCHES "^[0-9a-f]+(-dirty)?$")
  message(FATAL_ERROR "FIRMWARE_REVISION is required; use tools/build_firmware.py")
endif()
set(PROJECT_VER "${FIRMWARE_PROFILE}-${FIRMWARE_REVISION}")
string(LENGTH "${PROJECT_VER}" VERSION_LENGTH)
if(VERSION_LENGTH GREATER 31)
  message(FATAL_ERROR "Firmware version exceeds ESP app descriptor length")
endif()
if(FIRMWARE_PROFILE STREQUAL "kc")
  add_compile_definitions(KC_TEST_BUILD=1)
else()
  add_compile_definitions(KC_TEST_BUILD=0)
endif()
if(FIRMWARE_PROFILE STREQUAL "battery")
  add_compile_definitions(BATTERY_TEST_BUILD=1)
else()
  add_compile_definitions(BATTERY_TEST_BUILD=0)
endif()
add_compile_definitions(FIRMWARE_PROFILE="${FIRMWARE_PROFILE}"
  FIRMWARE_VERSION="${PROJECT_VER}" FIRMWARE_BUILD_TAG="idf-caltest/${PROJECT_VER}")
# USB serial uses nonblocking writes in setup(); UART0 remains available for GPS.
add_compile_definitions(ARDUINO_USB_CDC_ON_BOOT=1 ARDUINO_USB_MODE=1)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(caltest)
